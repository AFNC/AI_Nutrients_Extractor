    <h2>Upload from Path or Capture from Camera</h2>

    <!-- Video stream for live camera -->
    <video id="video" width="320" height="240" autoplay></video>
    <canvas id="canvas" width="320" height="240" style="display:none;"></canvas>
    <button onclick="capture()">Capture Photo</button>

    <!-- Form for upload -->
    <form action="/upload" method="post" enctype="multipart/form-data" class="upload-box">
        <p><strong>Upload from device (image or PDF):</strong></p>
        <input type="file" name="file" accept="image/*,.pdf"><br><br>

        <!-- Hidden input for captured webcam image -->
        <input type="hidden" name="capturedImage" id="capturedImage">

        <button type="submit">Submit</button>
    </form>

    <script>
        const video = document.getElementById('video');
        const canvas = document.getElementById('canvas');
        const capturedInput = document.getElementById('capturedImage');

        // Get camera stream
        navigator.mediaDevices.getUserMedia({ video: true })
            .then(stream => {
                video.srcObject = stream;
            })
            .catch(err => {
                console.error("Camera access error:", err);
            });

        function capture() {
            const ctx = canvas.getContext('2d');
            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
            canvas.style.display = 'block';
            const dataUrl = canvas.toDataURL('image/png');
            capturedInput.value = dataUrl;
        }
    </script>
